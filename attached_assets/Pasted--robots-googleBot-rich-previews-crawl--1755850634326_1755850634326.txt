Бонус: JSON-LD за услугите (повишава шанс за rich results)

Ако в /services изброяваш няколко услуги на една страница, добави ItemList със Service елементи.
(Може и на самата страница, не в layout.)

// app/services/JsonLd.tsx
"use client";
import Script from "next/script";

export default function ServicesJsonLd() {
  const data = {
    "@context": "https://schema.org",
    "@type": "ItemList",
    "itemListElement": [
      {
        "@type": "ListItem",
        "position": 1,
        "item": {
          "@type": "Service",
          "name": "SEO Struktor™",
          "url": "https://www.pravdagency.eu/services#seo-struktor"
        }
      },
      {
        "@type": "ListItem",
        "position": 2,
        "item": {
          "@type": "Service",
          "name": "Trendlab™",
          "url": "https://www.pravdagency.eu/services#trendlab"
        }
      },
      {
        "@type": "ListItem",
        "position": 3,
        "item": {
          "@type": "Service",
          "name": "Clickstarter™",
          "url": "https://www.pravdagency.eu/services#clickstarter"
        }
      },
      {
        "@type": "ListItem",
        "position": 4,
        "item": {
          "@type": "Service",
          "name": "Clientomat™",
          "url": "https://www.pravdagency.eu/services#clientomat"
        }
      }
    ]
  };

  return (
    <Script id="jsonld-services" type="application/ld+json"
      dangerouslySetInnerHTML={{ __html: JSON.stringify(data) }} />
  );
}
